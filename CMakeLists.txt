cmake_minimum_required(VERSION 3.28)
set(CMAKE_CXX_STANDARD 23)
project(cheryl)

set(CHERYL_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
set(CHERYL_BUILD_ROOT "${CMAKE_CURRENT_BINARY_DIR}")

option(WARN "Enable compiler warnings [-Wall]" OFF)
option(CHERYL_SANDBOX_BUILD "Deprecated native GLFW null-platform selection without Gainput" OFF)
option(CHERYL_BUILD_NATIVE_GLFW "Build the native GLFW window/input integration" ON)
option(CHERYL_BUILD_OPENGL "Build the OpenGL backend" ON)
option(CHERYL_BUILD_TESTS "Build Cheryl's tests and acceptance tools" ON)
option(CHERYL_BUILD_ALL_TESTS "Include the selected aggregate test runner in the default build" OFF)
option(CHERYL_BUILD_ACCEPTANCE_TESTS "Include broader acceptance runners in the default build" OFF)
option(CHERYL_BUILD_CONSUMER_TESTS "Build the consumer and public-header checks" OFF)
option(CHERYL_BUILD_DEMO "Build the windowed demonstration application" ON)
if(WARN)
    set(GCC_COVERAGE_COMPILE_FLAGS "-Wall")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} ${GCC_COVERAGE_COMPILE_FLAGS}" )
endif(WARN)
if (NOT MSVC AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    message("Debug build detected: enabling UB Sanitizer.")
    add_compile_options("-fsanitize=address,undefined")
    add_link_options("-fsanitize=address,undefined")
endif()

include(cmake/CherylDependencies.cmake)
include(cmake/CherylModules.cmake)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-Wno-deprecated-declarations)
endif()
if(CHERYL_BUILD_TESTS)
    include(GoogleTest)
    enable_testing()
endif()

add_subdirectory(projects/engine)

if(CHERYL_BUILD_OPENGL AND NOT CHERYL_BUILD_NATIVE_GLFW)
    message(FATAL_ERROR "Cheryl OpenGL requires Native GLFW; enable CHERYL_BUILD_NATIVE_GLFW or disable CHERYL_BUILD_OPENGL")
endif()
if(CHERYL_BUILD_NATIVE_GLFW)
    add_subdirectory(projects/modules/platform/native-glfw)
endif()
if(CHERYL_BUILD_OPENGL)
    add_subdirectory(projects/modules/graphics/opengl)
endif()

if(CHERYL_BUILD_CONSUMER_TESTS)
    add_subdirectory(projects/engine/tests/consumer)
endif()
if(CHERYL_BUILD_TESTS)
    add_subdirectory(projects/tools/backward-cpp)
endif()
if(CHERYL_BUILD_DEMO AND CHERYL_BUILD_OPENGL AND CHERYL_NATIVE_INPUT)
    add_subdirectory(projects/apps/demo)
endif()

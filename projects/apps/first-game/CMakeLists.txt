cmake_minimum_required(VERSION 3.28)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cheryl_first_game LANGUAGES CXX)
endif()

# A copied example can obtain these definitions from its installed SDK instead.
include("${CMAKE_CURRENT_LIST_DIR}/../../../cmake/CherylOptions.cmake" OPTIONAL)
if(COMMAND cheryl_configure_first_game_options)
    cheryl_configure_first_game_options()
endif()
set(first_game_source_consumer OFF)
if(NOT TARGET Cheryl::OpenGL::Startup)
    if(CHERYL_FIRST_GAME_ENGINE_SOURCE)
        get_filename_component(first_game_engine_source "${CHERYL_FIRST_GAME_ENGINE_SOURCE}" REALPATH
            BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        if(NOT EXISTS "${first_game_engine_source}/projects/engine/CMakeLists.txt")
            message(FATAL_ERROR "CHERYL_FIRST_GAME_ENGINE_SOURCE must name a Cheryl Engine checkout")
        endif()
        if(TARGET Cheryl::Engine)
            message(FATAL_ERROR "Supply the complete Cheryl OpenGL startup composition before adding first-game")
        endif()
        block(SCOPE_FOR VARIABLES)
            set(CHERYL_BUILD_TESTS OFF)
            set(CHERYL_BUILD_CONSUMER_TESTS OFF)
            set(CHERYL_BUILD_DEMO OFF)
            set(CHERYL_BUILD_FIRST_GAME OFF)
            set(CHERYL_BUILD_DOCS OFF)
            set(CHERYL_BUILD_SDK OFF)
            set(CHERYL_BUILD_DEBUG_TERMINAL_LINUX OFF)
            set(CHERYL_BUILD_UI_TGUI OFF)
            set(CHERYL_BUILD_UI_RMLUI OFF)
            set(CHERYL_BUILD_AUDIO_MINIAUDIO OFF)
            set(CHERYL_BUILD_WORKER_TOPOLOGY_HWLOC OFF)
            set(CHERYL_BUILD_WORKER_TOPOLOGY_JSON OFF)
            set(CHERYL_BUILD_NATIVE_GLFW ON)
            set(CHERYL_BUILD_OPENGL ON)
            set(CHERYL_NATIVE_INPUT ON)
            set(CHERYL_NATIVE_NULL_PLATFORM OFF)
            add_subdirectory("${first_game_engine_source}" cheryl EXCLUDE_FROM_ALL)
        endblock()
        set(first_game_source_consumer ON)
    else()
        find_package(Cheryl 4 CONFIG REQUIRED)
    endif()
endif()

if(NOT COMMAND cheryl_configure_first_game_options AND Cheryl_DIR)
    include("${Cheryl_DIR}/CherylOptions.cmake" OPTIONAL)
endif()
if(COMMAND cheryl_configure_first_game_options)
    cheryl_configure_first_game_options()
endif()

foreach(first_game_dependency IN ITEMS Cheryl::Engine Cheryl::NativeGLFW Cheryl::OpenGL Cheryl::OpenGL::Startup)
    if(NOT TARGET ${first_game_dependency})
        message(FATAL_ERROR "first-game requires ${first_game_dependency}; select an SDK with Native GLFW, OpenGL and startup")
    endif()
endforeach()
get_target_property(first_game_native_input Cheryl::NativeGLFW CHERYL_NATIVE_INPUT)
if(NOT first_game_native_input)
    message(FATAL_ERROR "first-game requires Native GLFW with CHERYL_NATIVE_INPUT enabled")
endif()

if(NOT DEFINED TARGET_APP_FIRST_GAME)
    set(TARGET_APP_FIRST_GAME first-game)
endif()
if(NOT DEFINED OUTPUT_APP_FIRST_GAME)
    set(OUTPUT_APP_FIRST_GAME first-game)
endif()
set(first_game_output "${CMAKE_CURRENT_BINARY_DIR}")
if(DEFINED CHERYL_BUILD_ROOT)
    set(first_game_output "${CHERYL_BUILD_ROOT}")
endif()

add_executable(${TARGET_APP_FIRST_GAME} src/main.cpp src/first-game.cpp)
set_target_properties(${TARGET_APP_FIRST_GAME} PROPERTIES
    OUTPUT_NAME "${OUTPUT_APP_FIRST_GAME}"
    RUNTIME_OUTPUT_DIRECTORY "${first_game_output}")
target_compile_definitions(${TARGET_APP_FIRST_GAME} PRIVATE
    CHERYL_FIRST_GAME_ASSETS="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_link_libraries(${TARGET_APP_FIRST_GAME} PRIVATE
    Cheryl::Engine Cheryl::NativeGLFW Cheryl::OpenGL Cheryl::OpenGL::Startup)

# Source Debug Engine exports its sanitizer link requirements. Its directory-local
# instrumentation does not reach this outer application's translation units.
if(first_game_source_consumer AND NOT MSVC AND CMAKE_BUILD_TYPE STREQUAL "Debug")
    target_compile_options(${TARGET_APP_FIRST_GAME} PRIVATE -fsanitize=address,undefined)
endif()

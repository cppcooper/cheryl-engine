cmake_minimum_required(VERSION 3.25)
project(cheryl_magic_m1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(MAGIC_SANITIZERS "Enable address/undefined sanitizers for this private project" ON)
add_library(magic_compiler STATIC src/compiler.cpp)
target_include_directories(magic_compiler PUBLIC include)
target_compile_options(magic_compiler PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(MAGIC_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(magic_compiler PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(magic_compiler PUBLIC -fsanitize=address,undefined)
endif()
add_executable(magic_inspect src/main.cpp)
target_link_libraries(magic_inspect PRIVATE magic_compiler)
target_compile_options(magic_inspect PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(magic_checks tests/compiler.cpp)
target_link_libraries(magic_checks PRIVATE magic_compiler)
target_compile_options(magic_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(magic_caller_checks tests/caller.cpp)
target_link_libraries(magic_caller_checks PRIVATE magic_compiler)
target_compile_options(magic_caller_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME magic_contracts COMMAND magic_checks)
add_test(NAME magic_callers COMMAND magic_caller_checks)
target_compile_definitions(magic_checks PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
foreach(fixture S1 S2 S3 S4 S5 mixed-schools)
    add_test(NAME magic_inspect_${fixture} COMMAND magic_inspect "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/${fixture}.verse")
endforeach()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME magic_cli COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.py"
    "$<TARGET_FILE:magic_inspect>" "${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
set_tests_properties(magic_contracts magic_callers magic_cli PROPERTIES TIMEOUT 30)
foreach(fixture S1 S2 S3 S4 S5 mixed-schools)
    set_tests_properties(magic_inspect_${fixture} PROPERTIES TIMEOUT 10)
endforeach()
add_library(magic_reference_host STATIC src/reference-host.cpp)
target_link_libraries(magic_reference_host PUBLIC magic_compiler)
target_compile_options(magic_reference_host PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(magic_host_checks tests/reference-host.cpp)
target_link_libraries(magic_host_checks PRIVATE magic_reference_host)
target_compile_options(magic_host_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(magic_host_checks PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_executable(magic_reference_demo src/reference-demo.cpp)
target_link_libraries(magic_reference_demo PRIVATE magic_reference_host)
target_compile_options(magic_reference_demo PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(magic_reference_demo PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_test(NAME magic_host_contracts COMMAND magic_host_checks)
add_test(NAME magic_reference_caller COMMAND magic_reference_demo)
set_tests_properties(magic_host_contracts magic_reference_caller PROPERTIES TIMEOUT 30)
add_executable(magic_engraving_checks tests/engraving.cpp)
target_link_libraries(magic_engraving_checks PRIVATE magic_reference_host)
target_compile_options(magic_engraving_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(magic_engraving_checks PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_executable(magic_engraving_demo src/engraving-demo.cpp)
target_link_libraries(magic_engraving_demo PRIVATE magic_reference_host)
target_compile_options(magic_engraving_demo PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME magic_engraving_contracts COMMAND magic_engraving_checks)
add_test(NAME magic_engraving_caller COMMAND magic_engraving_demo)
set_tests_properties(magic_engraving_contracts magic_engraving_caller PROPERTIES TIMEOUT 30)
add_library(magic_learning STATIC src/learning.cpp)
target_link_libraries(magic_learning PUBLIC magic_compiler)
target_compile_options(magic_learning PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_executable(magic_learning_checks tests/learning.cpp)
target_link_libraries(magic_learning_checks PRIVATE magic_learning magic_reference_host)
target_compile_options(magic_learning_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(magic_learning_checks PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_executable(magic_learning_demo src/learning-demo.cpp)
target_link_libraries(magic_learning_demo PRIVATE magic_learning magic_reference_host)
target_compile_options(magic_learning_demo PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(magic_learning_demo PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_test(NAME magic_learning_contracts COMMAND magic_learning_checks)
add_test(NAME magic_learning_caller COMMAND magic_learning_demo)
set_tests_properties(magic_learning_contracts magic_learning_caller PROPERTIES TIMEOUT 30)

target_link_libraries(magic_annular_checks PRIVATE magic_learning magic_reference_host)
target_compile_options(magic_annular_checks PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(magic_annular_checks PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")
add_executable(magic_annular_demo src/annular-demo.cpp)

target_compile_options(magic_annular_demo PRIVATE -Wall -Wextra -Wpedantic -Werror)
target_compile_definitions(magic_annular_demo PRIVATE MAGIC_FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/fixtures")

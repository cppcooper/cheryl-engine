# The aggregate is explicitly selectable; owners supply their own cases below.
if(CHERYL_BUILD_ALL_TESTS)
    add_executable(all-tests)
else()
    add_executable(all-tests EXCLUDE_FROM_ALL)
endif()
set_target_properties(all-tests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
target_link_libraries(all-tests PRIVATE cheryl_test_support)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_link_options(all-tests PRIVATE -rdynamic)
endif()
if(CHERYL_BUILD_ALL_TESTS)
    gtest_discover_tests(all-tests TEST_PREFIX "all-tests." WORKING_DIRECTORY "${CHERYL_BUILD_ROOT}")
endif()

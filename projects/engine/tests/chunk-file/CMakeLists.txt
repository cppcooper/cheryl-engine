cmake_minimum_required(VERSION 3.28)
project(cheryl_chunk_file_checks LANGUAGES CXX)
# Standalone opt-in harness: exercise production code without graphics/audio dependencies.
get_filename_component(CHERYL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CHERYL_CHUNK_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
if(CHERYL_CHUNK_SANITIZERS)
    if(NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "Chunk sanitizer harness requires GCC or Clang")
    endif()
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_library(cheryl_chunk_file STATIC
    "${CHERYL_ROOT}/projects/engine/src/core/resources/fileio/chunk-file.cpp"
    "${CHERYL_ROOT}/projects/engine/src/internal/exceptions.cpp")
target_include_directories(cheryl_chunk_file PUBLIC "${CHERYL_ROOT}/projects/engine/include")
target_include_directories(cheryl_chunk_file PRIVATE "${CHERYL_ROOT}/extern/backward-cpp")
target_compile_definitions(cheryl_chunk_file PRIVATE BACKWARD_HAS_DW=0 BACKWARD_HAS_BFD=0 BACKWARD_HAS_UNWIND=1)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cheryl_chunk_file PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${CHERYL_ROOT}/extern/googletest" "${CMAKE_CURRENT_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
add_executable(cheryl_chunk_file_tests tests.cpp)
target_include_directories(cheryl_chunk_file_tests PRIVATE "${CHERYL_ROOT}/projects/engine/src/core/resources/fileio")
target_link_libraries(cheryl_chunk_file_tests PRIVATE cheryl_chunk_file GTest::gtest_main)
add_executable(cheryl_chunk_file_example example.cpp)
target_link_libraries(cheryl_chunk_file_example PRIVATE cheryl_chunk_file)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(cheryl_chunk_file_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
    target_compile_options(cheryl_chunk_file_example PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME chunk_file COMMAND cheryl_chunk_file_tests)
add_test(NAME chunk_file_example COMMAND cheryl_chunk_file_example)

cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cheryl_opengl_consumer LANGUAGES CXX)
endif()
if(NOT TARGET Cheryl::OpenGL)
    block(SCOPE_FOR VARIABLES)
        set(CHERYL_BUILD_TESTS OFF)
        set(CHERYL_BUILD_CONSUMER_TESTS OFF)
        set(CHERYL_BUILD_DEMO OFF)
        set(CHERYL_BUILD_NATIVE_GLFW ON)
        set(CHERYL_BUILD_OPENGL ON)
        set(CHERYL_BUILD_UI_TGUI OFF)
        set(CHERYL_BUILD_UI_RMLUI OFF)
        add_subdirectory("${CHERYL_REPOSITORY_ROOT}" cheryl EXCLUDE_FROM_ALL)
    endblock()
endif()
include("${CHERYL_REPOSITORY_ROOT}/cmake/CherylConsumers.cmake")
file(GLOB_RECURSE consumer_sources CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_executable(${TARGET_TEST_CONSUMER_MODULE_OPENGL} ${consumer_sources})
set_target_properties(${TARGET_TEST_CONSUMER_MODULE_OPENGL} PROPERTIES OUTPUT_NAME "${OUTPUT_TEST_CONSUMER_MODULE_OPENGL}")
if(DEFINED CHERYL_BUILD_ROOT)
    set_target_properties(${TARGET_TEST_CONSUMER_MODULE_OPENGL} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
endif()
target_link_libraries(${TARGET_TEST_CONSUMER_MODULE_OPENGL} PRIVATE ${LINKAGE_PRIVATE_TEST_CONSUMER_MODULE_OPENGL})
add_subdirectory(headers)
add_dependencies(${TARGET_TEST_CONSUMER_MODULE_OPENGL} ${TARGET_TEST_CONSUMER_MODULE_OPENGL_HEADERS})

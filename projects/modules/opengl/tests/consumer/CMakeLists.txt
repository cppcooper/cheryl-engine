cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cheryl_opengl_consumer LANGUAGES CXX)
endif()
if(NOT TARGET Cheryl::OpenGL)
    if(NOT CHERYL_ENGINE_SOURCE)
        message(FATAL_ERROR "Supply CHERYL_ENGINE_SOURCE with the Cheryl checkout")
    endif()
    block(SCOPE_FOR VARIABLES)
        set(CHERYL_BUILD_TESTS OFF)
        set(CHERYL_BUILD_CONSUMER_TESTS OFF)
        set(CHERYL_BUILD_DEMO OFF)
        set(CHERYL_BUILD_NATIVE_GLFW ON)
        set(CHERYL_BUILD_OPENGL ON)
        add_subdirectory("${CHERYL_ENGINE_SOURCE}" cheryl EXCLUDE_FROM_ALL)
    endblock()
endif()
if(CHERYL_ENGINE_SOURCE)
    include("${CHERYL_ENGINE_SOURCE}/cmake/CherylConsumers.cmake")
else()
    include("${CHERYL_REPOSITORY_ROOT}/cmake/CherylConsumers.cmake")
endif()
add_executable(cheryl-opengl-consumer src/main.cpp)
if(DEFINED CHERYL_BUILD_ROOT)
    set_target_properties(cheryl-opengl-consumer PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
endif()
target_link_libraries(cheryl-opengl-consumer PRIVATE Cheryl::OpenGL)
add_subdirectory(headers)
add_dependencies(cheryl-opengl-consumer cheryl-opengl-consumer-headers)

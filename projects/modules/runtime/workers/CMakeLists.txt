cmake_minimum_required(VERSION 3.28)
include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/CherylModules.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cheryl_workers_module VERSION ${VERSION_PROJECT_MODULE_WORKERS} LANGUAGES C CXX)
endif()
if(TARGET Cheryl::Workers)
    return()
endif()
cheryl_configure_module_options()
cheryl_prepare_module()
cheryl_option(CHERYL_BUILD_WORKER_TOPOLOGY_HWLOC "Include native worker topology discovery" ON)
cheryl_option(CHERYL_BUILD_WORKER_TOPOLOGY_JSON "Include worker configuration loading" ON)

set(worker_sources)
if(CHERYL_BUILD_WORKER_TOPOLOGY_HWLOC)
    include(cmake/Hwloc.cmake)
    list(APPEND worker_sources src/hwloc.cpp)
    if(CHERYL_BUILD_TESTS)
        list(APPEND worker_sources src/test-access.cpp)
    endif()
endif()
if(CHERYL_BUILD_WORKER_TOPOLOGY_JSON)
    list(APPEND worker_sources src/configuration.cpp)
endif()
if(NOT worker_sources)
    message(FATAL_ERROR "Workers requires JSON configuration or hwloc topology support")
endif()
add_library(${TARGET_LIB_MODULE_WORKERS} ${CHERYL_LIBRARY_TYPE} ${worker_sources})
add_library(Cheryl::Workers ALIAS ${TARGET_LIB_MODULE_WORKERS})
cheryl_register_public_component(${TARGET_LIB_MODULE_WORKERS} FEATURE CHERYL_HAS_WORKERS)
set_target_properties(${TARGET_LIB_MODULE_WORKERS} PROPERTIES
    VERSION "${VERSION_PROJECT_MODULE_WORKERS}" OUTPUT_NAME "${OUTPUT_LIB_MODULE_WORKERS}"
    ARCHIVE_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
cheryl_configure_library(${TARGET_LIB_MODULE_WORKERS} WORKERS)
target_include_directories(${TARGET_LIB_MODULE_WORKERS} PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/cheryl>")
target_link_libraries(${TARGET_LIB_MODULE_WORKERS} PUBLIC Cheryl::Engine)
if(CHERYL_BUILD_WORKER_TOPOLOGY_HWLOC)
    add_library(Cheryl::WorkerTopology::Hwloc ALIAS ${TARGET_LIB_MODULE_WORKERS})
    target_compile_definitions(${TARGET_LIB_MODULE_WORKERS} PUBLIC CHERYL_HAS_SUPPORT_WORKER_TOPOLOGY_HWLOC=1)
    target_link_libraries(${TARGET_LIB_MODULE_WORKERS} PRIVATE ${cheryl_hwloc_target})
    cheryl_shared_dependency(${cheryl_hwloc_target} OWNER ${TARGET_LIB_MODULE_WORKERS})
endif()
if(CHERYL_BUILD_WORKER_TOPOLOGY_JSON)
    add_library(Cheryl::WorkerTopology::Json ALIAS ${TARGET_LIB_MODULE_WORKERS})
    target_compile_definitions(${TARGET_LIB_MODULE_WORKERS} PUBLIC CHERYL_HAS_SUPPORT_WORKER_TOPOLOGY_JSON=1)
    target_include_directories(${TARGET_LIB_MODULE_WORKERS} PRIVATE "${CHERYL_REPOSITORY_ROOT}/extern/json/single_include")
endif()
if(CHERYL_BUILD_WORKER_TOPOLOGY_HWLOC AND CHERYL_BUILD_WORKER_TOPOLOGY_JSON)
    add_library(${TARGET_LIB_ENGINE_S_WORKER_TOPOLOGY_EXAMPLE} STATIC EXCLUDE_FROM_ALL examples/shared-pool.cpp)
    set_target_properties(${TARGET_LIB_ENGINE_S_WORKER_TOPOLOGY_EXAMPLE} PROPERTIES
        OUTPUT_NAME "${OUTPUT_LIB_ENGINE_S_WORKER_TOPOLOGY_EXAMPLE}" ARCHIVE_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
    target_include_directories(${TARGET_LIB_ENGINE_S_WORKER_TOPOLOGY_EXAMPLE} PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/examples")
    target_link_libraries(${TARGET_LIB_ENGINE_S_WORKER_TOPOLOGY_EXAMPLE} PUBLIC Cheryl::Workers)
endif()
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cheryl_select_public_component(Cheryl::Workers)
    if(CHERYL_BUILD_TESTS)
        enable_testing()
        add_subdirectory(tests)
    endif()
endif()

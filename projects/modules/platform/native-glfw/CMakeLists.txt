cmake_minimum_required(VERSION 3.28)

include("${CMAKE_CURRENT_LIST_DIR}/../../../../cmake/CherylModules.cmake")
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cheryl_native_glfw_module VERSION ${VERSION_PROJECT_MODULE_NATIVE_GLFW} LANGUAGES C CXX)
endif()
if(TARGET Cheryl::NativeGLFW)
    return()
endif()
cheryl_configure_module_options()
cheryl_prepare_module()

cheryl_configure_native_options()

if(UNIX AND NOT APPLE)
    option(GAINPUT_ENABLE_HID "Include native HID controller support" OFF)
endif()
set(cheryl_native_hid OFF)
set(cheryl_hidapi_owned OFF)
if(CHERYL_NATIVE_INPUT AND (WIN32 OR GAINPUT_ENABLE_HID))
    set(cheryl_native_hid ON)
    include(cmake/Hidapi.cmake)
    cheryl_prepare_native_hidapi(cheryl_hidapi_owned)
endif()

set(cheryl_glfw_owned OFF)
if(NOT TARGET glfw)
    if(DEFINED GLFW_LIBRARY_TYPE AND GLFW_LIBRARY_TYPE AND NOT GLFW_LIBRARY_TYPE STREQUAL "SHARED")
        message(FATAL_ERROR "NativeGLFW requires GLFW_LIBRARY_TYPE=SHARED; supplied dependency settings are not rewritten")
    endif()
    block(SCOPE_FOR VARIABLES)
        set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
        set(GLFW_BUILD_DOCS OFF)
        set(BUILD_SHARED_LIBS ON)
        set(GLFW_LIBRARY_TYPE SHARED)
        if(CHERYL_NATIVE_NULL_PLATFORM)
            set(GLFW_BUILD_X11 OFF)
            set(GLFW_BUILD_WAYLAND OFF)
        endif()
        add_subdirectory("${CHERYL_REPOSITORY_ROOT}/extern/glfw" extern/glfw EXCLUDE_FROM_ALL)
    endblock()
    set_target_properties(glfw PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
    set(cheryl_glfw_owned ON)
endif()
if(cheryl_hidapi_owned)
    get_target_property(cheryl_hidapi_target hidapi::hidapi ALIASED_TARGET)
    if(NOT cheryl_hidapi_target)
        set(cheryl_hidapi_target hidapi::hidapi)
    endif()
    set_target_properties(${cheryl_hidapi_target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
endif()

file(GLOB_RECURSE cheryl_native_sources CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
if(NOT CHERYL_NATIVE_INPUT)
    list(FILTER cheryl_native_sources EXCLUDE REGEX "/src/core/controls/")
endif()
if(CHERYL_NATIVE_INPUT)
    if(UNIX AND NOT APPLE AND NOT TARGET X11::X11)
        find_package(X11 REQUIRED GLOBAL)
    endif()
    set(cheryl_gainput_added OFF)
    set(cheryl_gainput_target "")
    foreach(candidate IN LISTS LINKAGE_PUBLIC_LIB_MODULE_NATIVE_GLFW_INPUT_GAINPUT_CANDIDATES)
        if(TARGET ${candidate})
            set(cheryl_gainput_target ${candidate})
            break()
        endif()
    endforeach()
    if(NOT cheryl_gainput_target)
        block(SCOPE_FOR VARIABLES)
            set(CMAKE_POLICY_DEFAULT_CMP0077 NEW)
            set(GAINPUT_SAMPLES OFF)
            set(GAINPUT_TESTS OFF)
            set(BUILD_SHARED_LIBS OFF)
            add_subdirectory("${CHERYL_REPOSITORY_ROOT}/extern/gainput" extern/gainput EXCLUDE_FROM_ALL)
        endblock()
        set(cheryl_gainput_added ON)
        foreach(candidate IN LISTS LINKAGE_PUBLIC_LIB_MODULE_NATIVE_GLFW_INPUT_GAINPUT_CANDIDATES)
            if(TARGET ${candidate})
                set(cheryl_gainput_target ${candidate})
                break()
            endif()
        endforeach()
    endif()
    if(NOT cheryl_gainput_target)
        message(FATAL_ERROR "Gainput must provide one of: ${LINKAGE_PUBLIC_LIB_MODULE_NATIVE_GLFW_INPUT_GAINPUT_CANDIDATES}")
    endif()
    if(cheryl_gainput_added)
        get_target_property(cheryl_gainput_concrete ${cheryl_gainput_target} ALIASED_TARGET)
        if(NOT cheryl_gainput_concrete)
            set(cheryl_gainput_concrete ${cheryl_gainput_target})
        endif()
        # The legacy dependency uses directory-local includes instead of exporting
        # them. Repair its owned target; modern/supplied targets export their own.
        if(NOT TARGET gainput::gainput)
            target_include_directories(${cheryl_gainput_concrete} SYSTEM INTERFACE
                "$<BUILD_INTERFACE:${CHERYL_REPOSITORY_ROOT}/extern/gainput/lib/include>")
        endif()
        if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
            target_compile_options(${cheryl_gainput_concrete} PRIVATE -Wno-implicit-fallthrough -Wno-unused-parameter)
        endif()
    endif()
endif()

add_library(${TARGET_LIB_MODULE_NATIVE_GLFW} ${CHERYL_LIBRARY_TYPE} ${cheryl_native_sources})
add_library(Cheryl::NativeGLFW ALIAS ${TARGET_LIB_MODULE_NATIVE_GLFW})
cheryl_register_public_component(${TARGET_LIB_MODULE_NATIVE_GLFW} FEATURE CHERYL_HAS_NATIVE_GLFW)
set_target_properties(${TARGET_LIB_MODULE_NATIVE_GLFW} PROPERTIES
    VERSION "${VERSION_PROJECT_MODULE_NATIVE_GLFW}"
    OUTPUT_NAME "${OUTPUT_LIB_MODULE_NATIVE_GLFW}"
    ARCHIVE_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}"
    CHERYL_NATIVE_INPUT "${CHERYL_NATIVE_INPUT}")
cheryl_configure_library(${TARGET_LIB_MODULE_NATIVE_GLFW} NATIVE_GLFW)
if(cheryl_glfw_owned)
    cheryl_shared_dependency(glfw OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW} OWNED)
else()
    cheryl_shared_dependency(glfw OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW})
endif()
target_include_directories(${TARGET_LIB_MODULE_NATIVE_GLFW} PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
    "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include/cheryl>")
target_compile_definitions(${TARGET_LIB_MODULE_NATIVE_GLFW} PUBLIC CHERYL_NATIVE_INPUT=$<BOOL:${CHERYL_NATIVE_INPUT}>)
target_link_libraries(${TARGET_LIB_MODULE_NATIVE_GLFW}
    PUBLIC ${LINKAGE_PUBLIC_LIB_MODULE_NATIVE_GLFW}
    PRIVATE ${LINKAGE_PRIVATE_LIB_MODULE_NATIVE_GLFW})
if(CHERYL_NATIVE_INPUT)
    target_link_libraries(${TARGET_LIB_MODULE_NATIVE_GLFW} PRIVATE ${cheryl_gainput_target})
    if(cheryl_native_hid)
        if(cheryl_hidapi_owned)
            cheryl_shared_dependency(hidapi::hidapi OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW} OWNED)
        else()
            cheryl_shared_dependency(hidapi::hidapi OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW})
        endif()
        if(UNIX AND NOT APPLE)
            find_package(PkgConfig REQUIRED)
            pkg_check_modules(CHERYL_NATIVE_UDEV REQUIRED IMPORTED_TARGET GLOBAL libudev)
            cheryl_shared_dependency(PkgConfig::CHERYL_NATIVE_UDEV OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW} SYSTEM)
            target_link_libraries(${TARGET_LIB_MODULE_NATIVE_GLFW} PRIVATE PkgConfig::CHERYL_NATIVE_UDEV)
        endif()
    endif()
    if(cheryl_gainput_added)
        cheryl_private_dependency(${cheryl_gainput_target} OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW} OWNED)
    else()
        cheryl_private_dependency(${cheryl_gainput_target} OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW})
    endif()
    if(UNIX AND NOT APPLE)
        cheryl_shared_dependency(X11::X11 OWNER ${TARGET_LIB_MODULE_NATIVE_GLFW} SYSTEM)
        target_link_libraries(${TARGET_LIB_MODULE_NATIVE_GLFW} PRIVATE ${LINKAGE_PRIVATE_LIB_MODULE_NATIVE_GLFW_INPUT_LINUX})
    endif()
endif()

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cheryl_select_public_component(${TARGET_LIB_MODULE_NATIVE_GLFW})
endif()

if(CHERYL_BUILD_TESTS)
    target_sources(${TARGET_LIB_MODULE_NATIVE_GLFW} PRIVATE tests/support/src/native-access.cpp)
    target_include_directories(${TARGET_LIB_MODULE_NATIVE_GLFW} PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src")
    target_compile_definitions(${TARGET_LIB_MODULE_NATIVE_GLFW} PRIVATE CHERYL_NATIVE_TEST_SUPPORT=1)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux" AND GLFW_BUILD_X11)
        target_compile_definitions(${TARGET_LIB_MODULE_NATIVE_GLFW} PRIVATE CHERYL_NATIVE_X11_TESTS=1)
    endif()
    enable_testing()
    add_subdirectory(tests)
endif()
if(CHERYL_BUILD_CONSUMER_TESTS)
    add_subdirectory(tests/consumer)
endif()

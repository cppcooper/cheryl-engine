cmake_minimum_required(VERSION 3.28)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(cheryl_ui_rmlui_consumer LANGUAGES CXX)
endif()
if(NOT TARGET Cheryl::UI::RmlUi)
    block(SCOPE_FOR VARIABLES)
        set(CHERYL_BUILD_TESTS OFF)
        set(CHERYL_BUILD_CONSUMER_TESTS OFF)
        set(CHERYL_BUILD_ROOT "${CMAKE_BINARY_DIR}")
        add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." adapter EXCLUDE_FROM_ALL)
    endblock()
endif()
include("${CHERYL_REPOSITORY_ROOT}/cmake/CherylConsumers.cmake")
file(GLOB_RECURSE consumer_sources CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")
add_executable(${TARGET_TEST_CONSUMER_MODULE_UI_RMLUI} ${consumer_sources})
set_target_properties(${TARGET_TEST_CONSUMER_MODULE_UI_RMLUI} PROPERTIES OUTPUT_NAME "${OUTPUT_TEST_CONSUMER_MODULE_UI_RMLUI}")
if(DEFINED CHERYL_BUILD_ROOT)
    set_target_properties(${TARGET_TEST_CONSUMER_MODULE_UI_RMLUI} PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY "${CHERYL_BUILD_ROOT}")
endif()
target_link_libraries(${TARGET_TEST_CONSUMER_MODULE_UI_RMLUI} PRIVATE ${LINKAGE_PRIVATE_TEST_CONSUMER_MODULE_UI_RMLUI})
include("${CMAKE_CURRENT_LIST_DIR}/../../cmake/TestFont.cmake")
cheryl_rmlui_test_font(test_font)
target_compile_definitions(${TARGET_TEST_CONSUMER_MODULE_UI_RMLUI} PRIVATE CHERYL_RMLUI_TEST_FONT="${test_font}")
add_subdirectory(headers)
add_dependencies(${TARGET_TEST_CONSUMER_MODULE_UI_RMLUI} ${TARGET_TEST_CONSUMER_MODULE_UI_RMLUI_HEADERS})

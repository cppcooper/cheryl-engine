set(sites_procgen "${CMAKE_CURRENT_SOURCE_DIR}/../../../procedural-generation")
set(LIVING_WORLD_SITES_CORE_TARGET "" CACHE STRING "Existing private target owning maze.cpp and random.cpp, or empty for a new core")
if(LIVING_WORLD_SITES_CORE_TARGET)
    if(NOT TARGET "${LIVING_WORLD_SITES_CORE_TARGET}")
        message(FATAL_ERROR "LIVING_WORLD_SITES_CORE_TARGET must name an existing maze/random owner")
    endif()
else()
    if(TARGET world-cave-private OR TARGET procgen-private)
        message(FATAL_ERROR "Select the existing maze/random owner with LIVING_WORLD_SITES_CORE_TARGET; do not duplicate these sources")
    endif()
    add_library(living-world-sites-core STATIC "${sites_procgen}/maze.cpp" "${sites_procgen}/random.cpp")
    target_include_directories(living-world-sites-core PUBLIC "${sites_procgen}")
    target_compile_features(living-world-sites-core PUBLIC cxx_std_23)
    set_target_properties(living-world-sites-core PROPERTIES CXX_EXTENSIONS OFF)
    set(LIVING_WORLD_SITES_CORE_TARGET living-world-sites-core)
endif()
if(LIVING_WORLD_SITES_CORE_TARGET STREQUAL "procgen-private")
    message(FATAL_ERROR "The standalone all-producer target already owns elevation/selector; select a maze/random-only private owner")
endif()

add_library(living-world-sites-producer STATIC "${sites_procgen}/elevation.cpp" "${sites_procgen}/settlement-sites.cpp")
target_include_directories(living-world-sites-producer PUBLIC "${sites_procgen}")
target_link_libraries(living-world-sites-producer PUBLIC "${LIVING_WORLD_SITES_CORE_TARGET}")
add_library(living-world-sites-adapter STATIC sites.cpp)
target_include_directories(living-world-sites-adapter PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(living-world-sites-adapter PUBLIC living-world-sites-producer)
foreach(sites_target IN ITEMS living-world-sites-producer living-world-sites-adapter)
    target_compile_features(${sites_target} PUBLIC cxx_std_23)
    set_target_properties(${sites_target} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${sites_target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
    endif()
endforeach()
if(TARGET living-world-sites-core AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(living-world-sites-core PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
endif()

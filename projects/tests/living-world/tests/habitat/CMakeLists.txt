if(NOT TARGET living-world-habitat OR NOT TARGET Cheryl::Engine)
    message(FATAL_ERROR "Add habitat and the existing shared Engine before habitat consumers")
endif()
get_target_property(habitat_engine_kind Cheryl::Engine TYPE)
if(NOT habitat_engine_kind STREQUAL "SHARED_LIBRARY" OR TARGET Cheryl::MapStorageCPU)
    message(FATAL_ERROR "Habitat consumer requires one shared Engine and no standalone storage producer")
endif()
add_library(living-world-habitat-fixtures STATIC fixtures.cpp)
target_include_directories(living-world-habitat-fixtures PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(living-world-habitat-fixtures PUBLIC living-world-habitat)
add_executable(tests-habitat-selector selector.cpp)
add_executable(tests-habitat tests.cpp)
add_executable(tests-habitat-allocation allocation.cpp)
add_executable(living-world-habitat-consumer consumer.cpp)
foreach(habitat_target IN ITEMS tests-habitat-selector tests-habitat tests-habitat-allocation living-world-habitat-consumer)
    target_link_libraries(${habitat_target} PRIVATE living-world-habitat-fixtures)
endforeach()
target_link_libraries(living-world-habitat-consumer PRIVATE Cheryl::Engine)
foreach(habitat_target IN ITEMS living-world-habitat-fixtures tests-habitat-selector tests-habitat tests-habitat-allocation living-world-habitat-consumer)
    target_compile_features(${habitat_target} PRIVATE cxx_std_23)
    set_target_properties(${habitat_target} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${habitat_target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
    endif()
endforeach()
foreach(habitat_case IN ITEMS literal invalid budgets ordering)
    add_test(NAME "living-world.habitat.selector.${habitat_case}" COMMAND tests-habitat-selector "${habitat_case}")
    set_tests_properties("living-world.habitat.selector.${habitat_case}" PROPERTIES TIMEOUT 30 LABELS "living-world;habitat;selector")
endforeach()
foreach(habitat_case IN ITEMS literal generated invalid budgets water excavation fences revisions lifetime)
    add_test(NAME "living-world.habitat.${habitat_case}" COMMAND tests-habitat "${habitat_case}")
    set_tests_properties("living-world.habitat.${habitat_case}" PROPERTIES TIMEOUT 30 LABELS "living-world;habitat")
endforeach()
add_test(NAME living-world.habitat.allocation COMMAND tests-habitat-allocation)
add_test(NAME living-world.habitat.consumer COMMAND living-world-habitat-consumer)
set_tests_properties(living-world.habitat.allocation living-world.habitat.consumer PROPERTIES TIMEOUT 30 LABELS "living-world;habitat")

if(NOT TARGET living-world-sites-adapter OR NOT TARGET Cheryl::Engine)
    message(FATAL_ERROR "Add the sites leaf and the existing Engine before its consumers")
endif()
get_target_property(sites_engine_kind Cheryl::Engine TYPE)
if(NOT sites_engine_kind STREQUAL "SHARED_LIBRARY" OR TARGET Cheryl::MapStorageCPU)
    message(FATAL_ERROR "Sites consumers require one normal shared Engine and no standalone storage producer")
endif()
add_library(living-world-sites-fixtures STATIC fixtures.cpp)
target_link_libraries(living-world-sites-fixtures PUBLIC living-world-sites-adapter)
target_include_directories(living-world-sites-fixtures PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(tests-world-sites tests.cpp)
add_executable(living-world-sites-consumer consumer.cpp)
foreach(sites_consumer IN ITEMS tests-world-sites living-world-sites-consumer)
    target_link_libraries(${sites_consumer} PRIVATE living-world-sites-fixtures Cheryl::Engine)
endforeach()
foreach(sites_target IN ITEMS living-world-sites-fixtures tests-world-sites living-world-sites-consumer)
    target_compile_features(${sites_target} PRIVATE cxx_std_23)
    set_target_properties(${sites_target} PROPERTIES CXX_EXTENSIONS OFF)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${sites_target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wsign-conversion -Werror)
    endif()
endforeach()
foreach(sites_case IN ITEMS literal generated water excavation invalid budgets lifetime fences retention)
    add_test(NAME "living-world.sites.${sites_case}" COMMAND tests-world-sites "${sites_case}")
    set_tests_properties("living-world.sites.${sites_case}" PROPERTIES TIMEOUT 30 LABELS "living-world;sites")
endforeach()
add_test(NAME living-world.sites.consumer COMMAND living-world-sites-consumer)
set_tests_properties(living-world.sites.consumer PROPERTIES TIMEOUT 30 LABELS "living-world;sites")
